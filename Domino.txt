<script>
const ROWS = 8;
const COLS = 4;
let state = {
  score1:0, score2:0,
  name1:"محمد", name2:"ئەحمەد",
  cells1: Array.from({length:ROWS},()=>({vals:["","","",""], dots:[false,false,false,false]})),
  cells2: Array.from({length:ROWS},()=>({vals:["","","",""], dots:[false,false,false,false]})),
  dark:false
};

function save(){
  try{ localStorage.setItem('dominoState', JSON.stringify(state)); }catch(e){}
}
function load(){
  try{
    const s = localStorage.getItem('dominoState');
    if(s){ state = Object.assign(state, JSON.parse(s)); }
  }catch(e){}
}

function addScore(player, amount){
  if(player===1){ state.score1 += amount; if(state.score1<0) state.score1=0; }
  else { state.score2 += amount; if(state.score2<0) state.score2=0; }
  render();
  save();
}

function resetAll(){
  if(!confirm("دڵنیایت لە سفرکردنەوەی هەموو خاڵەکان؟")) return;
  state.score1=0; state.score2=0;
  state.cells1 = Array.from({length:ROWS},()=>({vals:["","","",""], dots:[false,false,false,false]}));
  state.cells2 = Array.from({length:ROWS},()=>({vals:["","","",""], dots:[false,false,false,false]}));
  buildBoard();
  render();
  save();
}

function resetGroup(player){
  const pname = player===1 ? state.name1 : state.name2;
  if(!confirm("دڵنیایت لە سفرکردنەوەی خاڵەکانی " + pname + "؟")) return;
  if(player===1){
    state.score1 = 0;
    state.cells1 = Array.from({length:ROWS},()=>({vals:["","","",""], dots:[false,false,false,false]}));
  } else {
    state.score2 = 0;
    state.cells2 = Array.from({length:ROWS},()=>({vals:["","","",""], dots:[false,false,false,false]}));
  }
  buildBoard();
  render();
  save();
}

function toggleDark(){
  state.dark = !state.dark;
  document.body.classList.toggle('dark', state.dark);
  save();
}

function buildBoard(){
  const board = document.getElementById('board');
  board.innerHTML = "";
  const col1 = document.createElement('div');
  col1.className = 'board-col';
  const col2 = document.createElement('div');
  col2.className = 'board-col';

  for(let r=0;r<ROWS;r++){
    col1.appendChild(makeRowPanel('red', 1, r));
    col2.appendChild(makeRowPanel('blue', 2, r));
  }
  board.appendChild(col1);
  board.appendChild(col2);
}

function makeRowPanel(color, player, rowIndex){
  const panel = document.createElement('div');
  panel.className = 'row-panel ' + color;

  const top = document.createElement('div');
  top.className = 'row-top';
  const bottom = document.createElement('div');
  bottom.className = 'row-bottom';

  for(let c=0;c<COLS;c++){
    const input = document.createElement('input');
    input.className = 'cell-input';
    input.maxLength = 2;
    input.inputMode = 'numeric';
    const key = player===1?'cells1':'cells2';
    input.value = state[key][rowIndex].vals[c];
    input.addEventListener('input', (e)=>{
      let v = e.target.value.replace(/[^0-9]/g,'');
      state[key][rowIndex].vals[c] = v;
      e.target.value = v;
      save();
    });
    top.appendChild(input);

    const dot = document.createElement('div');
    dot.className = 'cell-dot' + (state[key][rowIndex].dots[c] ? ' active' : '');
    dot.addEventListener('click', ()=>{
      state[key][rowIndex].dots[c] = !state[key][rowIndex].dots[c];
      dot.classList.toggle('active');
      save();
    });
    bottom.appendChild(dot);
  }
  panel.appendChild(top);
  panel.appendChild(bottom);
  return panel;
}

function render(){
  document.getElementById('score1').textContent = state.score1;
  document.getElementById('score2').textContent = state.score2;
  document.getElementById('name1').value = state.name1;
  document.getElementById('name2').value = state.name2;
  document.getElementById('resetName1').textContent = state.name1;
  document.getElementById('resetName2').textContent = state.name2;
  document.body.classList.toggle('dark', state.dark);
}

document.getElementById('name1').addEventListener('input', e=>{
  state.name1 = e.target.value;
  document.getElementById('resetName1').textContent = state.name1;
  save();
});
document.getElementById('name2').addEventListener('input', e=>{
  state.name2 = e.target.value;
  document.getElementById('resetName2').textContent = state.name2;
  save();
});

load();
buildBoard();
render();
